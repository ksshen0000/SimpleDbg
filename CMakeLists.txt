cmake_minimum_required(VERSION 3.18)
project(SimpleDbg LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
    message(FATAL_ERROR "SimpleDbg requires Linux x86-64")
endif()
find_path(CAPSTONE_INCLUDE_DIR capstone/capstone.h REQUIRED)
find_library(CAPSTONE_LIBRARY capstone REQUIRED)
add_library(sdb_core STATIC sdb.cpp symbols.cpp process_maps.cpp execution.cpp state.cpp source_tools.cpp source_commands.cpp)
target_compile_features(sdb_core PUBLIC cxx_std_17)
target_compile_options(sdb_core PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(sdb_core PUBLIC ${CAPSTONE_INCLUDE_DIR})
target_link_libraries(sdb_core PUBLIC ${CAPSTONE_LIBRARY})
add_executable(sdb main.cpp)
target_compile_options(sdb PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(sdb PRIVATE sdb_core)
option(SDB_ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(SDB_ENABLE_COVERAGE "Enable GCC/Clang gcov instrumentation" OFF)
if(SDB_ENABLE_SANITIZERS)
    target_compile_options(sdb_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(sdb_core PUBLIC -fsanitize=address,undefined)
endif()
if(SDB_ENABLE_COVERAGE)
    target_compile_options(sdb_core PUBLIC --coverage -O0 -g)
    target_link_options(sdb_core PUBLIC --coverage)
endif()
include(CTest)
if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(sdb-fault main.cpp tests/ptrace_fault.cpp)
    target_link_libraries(sdb-fault PRIVATE sdb_core)
    target_link_options(sdb-fault PRIVATE -Wl,--wrap=ptrace -Wl,--wrap=process_vm_writev)
    target_compile_options(sdb-fault PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_debugger.py $<TARGET_FILE:sdb>)
    add_test(NAME features COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_features.py $<TARGET_FILE:sdb>)
    add_test(NAME failures COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_faults.py $<TARGET_FILE:sdb-fault>)
    set_tests_properties(integration features failures PROPERTIES TIMEOUT 240)
endif()
